# 脚本

`build.py`：预筛并出初判分段；`mkplan.py`：终稿转写入计划（需 bim 引擎与四个并列仓）。

## build.py

```python
import csv,json,glob,collections,sys,os
csv.field_size_limit(10**9)
BI='/home/user/book-index'; RUN='/home/user/scratch/run'; OUT='/home/user/scratch/a9'
SRC={'utokyo':'東京大学デジタルアーカイブ漢籍','ioc':'東洋文化研究所漢籍善本全文影像','shoryobu':'宮内庁書陵部図書寮文庫漢籍','keio':'慶應義塾図書館漢籍コレクション','arc':'立命館ARC古典籍漢籍','nagoya':'名古屋大学附属図書館漢籍'}
SITE={'utokyo':'utokyo-da','ioc':'ioc-shanben','shoryobu':'shoryobu','keio':'keio','arc':'arc-ritsumei','nagoya':'nagoya-u'}
bad=set(open('/mnt/project-files/a2-日本三馆/A2已判有疑Work.txt').read().split())
def load(wid):
    g=glob.glob(f'{BI}/Work/{wid[-3]}/{wid[-2]}/{wid[-1]}/{wid}*.json'); return json.load(open(g[0])) if g else None
def wdesc(wid,site):
    w=load(wid)
    if not w: return f'{wid}:缺'
    if w.get('merged_into'):
        k=w['merged_into']; w=load(k) or w; wid=k
    au='、'.join(f"{a.get('name','')}({a.get('dynasty','')}){a.get('role','')}" for a in (w.get('authors') or []) if isinstance(a,dict))
    jc=w.get('juan_count',{}).get('number','') if isinstance(w.get('juan_count'),dict) else ''
    ib='；'.join(f"{i.get('source','')}:{i.get('title_info','')}{i.get('author_info','')}" for i in (w.get('indexed_by') or [])[:2])
    n=sum(1 for x in (w.get('resources') or []) if x.get('id')==site)
    return f"[{wid}] {w.get('title','')} | {au} | 卷{jc} | {ib[:150]} | 已挂本站{n}"
drop=collections.Counter(); keep=[]; dedup=[]
for s in SRC:
    for r in csv.DictReader(open(f'{RUN}/{s}/matches.tsv'),delimiter='\t'):
        if r['tier']!='存疑': continue
        codes=set(r['reason_codes'].split(';')); cands=[c for c in r['candidates'].split(',') if c]
        why=None
        if r['has_image']!='1': why='K3无影像'
        elif r['already_linked']=='1' or r['excluded']: why='已挂/排除'
        elif not r['lib_author'].strip(): why='馆藏无责任者'
        elif 'GROUP_SPLIT' in codes or 'DUP_WORK_SUSPECT' in codes: why='疑重复Work交去重'; dedup.append((s,r))
        elif any(c in bad for c in cands): why='A2有疑Work'
        elif len(cands)!=1: why='多候选'
        drop[(s,why)]+=1
        if not why: r['s']=s; keep.append(r)
for k,v in sorted(drop.items(),key=str): print(k,v)
print(len(keep))
rows=[]
for r in keep:
    rows.append([r['s'],r['source_record_id'],r['reason_codes'],r['lib_title'],r['lib_author'],r['lib_version'],wdesc(r['candidates'].split(',')[0],SITE[r['s']])])
hdr=['馆','record_id','原因码','馆藏题名','馆藏责任者','馆藏版本','候选Work（[id] 题名|著者(朝代)|卷数|著录|已挂本站数）']
n=int(sys.argv[1]); size=(len(rows)+n-1)//n
for i in range(n):
    with open(f'{OUT}/段{i+1}.tsv','w') as f:
        w=csv.writer(f,delimiter='\t',lineterminator='\n'); w.writerow(['序号']+hdr)
        for k,x in enumerate(rows[i*size:(i+1)*size],1): w.writerow([k]+x)
with open(f'{OUT}/交去重_疑重复Work.tsv','w') as f:
    w=csv.writer(f,delimiter='\t',lineterminator='\n'); w.writerow(['馆','record_id','题名','责任者','原因码','候选Work'])
    for s,r in dedup: w.writerow([s,r['source_record_id'],r['lib_title'],r['lib_author'],r['reason_codes'],r['candidates']])
print(len(rows),size)

```

## mkplan.py

```python
import sys,os,csv,json,collections,importlib
sys.path.insert(0,'/home/user/book-index-manager/tools/馆藏资源挂接/match_engine')
import writeplan as WP, workindex
A='/home/user/scratch/a9'; DATA='/home/user/book_index_json/library_data'; BI='/home/user/book-index'
final=json.load(open(f'{A}/最终_无arc.json'))   # record_id -> {work_id, note, source}
res=workindex.Resources.scan(BI)
bysrc=collections.defaultdict(dict)
for rid,v in final.items(): bysrc[v['source']][rid]=v
os.makedirs(f'{A}/plan',exist_ok=True)
allplan=[];st=collections.Counter()
for s,m in bysrc.items():
    ad=importlib.import_module('adapters.'+s); recs=ad.load(DATA)
    byid={r['source_record_id']:r for r in recs}
    rows=[]
    for rid,v in m.items():
        r=byid.get(rid)
        if r is None: st['缺记录']+=1; continue
        rows.append(dict(rec=r,tier='确定',work_id=v['work_id'],codes=[],excluded='',already_linked=0,note=v['note']))
    plan,dropped,linked=WP.build(s,rows,res.existing_groups(s))
    for p in plan:
        n=m[p['record_id']]['note'].strip()
        if n: p['resource']['note']=n
    st[s]+=len(plan); st[s+'_dropped']+=len(dropped)
    d=f'{A}/plan/{s}'; os.makedirs(d,exist_ok=True)
    open(f'{d}/write_plan.jsonl','w').write(''.join(json.dumps(p,ensure_ascii=False)+'\n' for p in plan))
    allplan+=plan
open(f'{A}/plan/write_plan.jsonl','w').write(''.join(json.dumps(p,ensure_ascii=False)+'\n' for p in allplan))
print(st,len(allplan))

```
